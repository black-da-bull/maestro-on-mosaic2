SEM composite + release gate (Python)

# Song Excellence Matrix — source S0816 / CR-009 @ S2117
def seg_composite(scores, weights):           # scores 0–5, weights sum to 100
    pct = sum((scores[k]/5)*weights[k] for k in weights) / sum(weights.values())
    return round(pct*100, 1)

RELEASE_FLOOR = 97.5     # CR-009 supersedes legacy ≥70 (origin)
MINIMA = {"hook_strength":4, "lyric_integrity":3, "lyric_syllable_integrity":4, ...}

def gate(scores, weights):
    if any(scores[k] < m for k,m in MINIMA.items()): return "HOLD"
    c = seg_composite(scores, weights)
    return "RELEASE" if c >= RELEASE_FLOOR else ("REVISE" if c >= 60 else "REWORK")
