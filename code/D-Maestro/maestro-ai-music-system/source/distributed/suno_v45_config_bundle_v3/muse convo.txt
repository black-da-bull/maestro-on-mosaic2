import os, yaml, json, time, urllib.parse
from openai import ChatCompletion

# Load prompts
with open('prompts/core_prompt.md')  as f: core = f.read()
with open('prompts/session_shim.md') as f: shim = f.read()
config = yaml.safe_load(open('config/workflow_config.yaml'))

# Prepare API client
client = ChatCompletion()

def generate_variant(bpm_offset):
    # inject chaos BPM drift
    prompt = core.replace("Tempo: 80 BPM", f"Tempo: {80 + bpm_offset} BPM") + "\n" + shim
    return prompt

logs = []
for i in range(config['generation']['chaos_variants']):
    bpm_drift = ((-1)**i) * (i+1)  # e.g. -1, +2, -3 ...
    prompt = generate_variant(bpm_drift)
    resp = client.create(model="gpt-4o-mini", messages=[{"role":"system","content":prompt}])
    output = resp.choices[0].message.content
    timestamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    log_entry = {
        "timestamp": timestamp,
        "variant": i,
        "bpm_drift": bpm_drift,
        "prompt_version": "v2.0-core",
        "output": output
    }
    logs.append(log_entry)
    print(output)  # For CI logs

# Append to logfile
with open('logs/generation_logs.jsonl','a') as lf:
    for entry in logs:
        lf.write(json.dumps(entry)+"\n")
