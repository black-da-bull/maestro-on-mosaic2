#!/usr/bin/env python3
import sys, json, re, argparse
from pathlib import Path

ORDER = ["Summary","Theory","Voice","Crew_Tags","Road-Map","Lyrics","Style","Timbre","Performance","Exit"]
BUDGET_SUMMARY = 1000
BUDGET_MACRO = 5000

def read_text(p: Path) -> str:
    return p.read_text(encoding="utf-8")

def section_slices(txt: str):
    # returns {name: (start,end)}
    pat = re.compile(r'^\[([A-Za-z_\-]+)\]\s*$', flags=re.M)
    boundaries = [(m.group(1), m.start(), m.end()) for m in pat.finditer(txt)]
    sections = {}
    for i, (name, s, e) in enumerate(boundaries):
        start = e
        end = boundaries[i+1][1] if i+1 < len(boundaries) else len(txt)
        sections[name] = (start, end)
    return sections

def get_section(txt: str, name: str) -> str:
    secs = section_slices(txt)
    if name not in secs: return ""
    s, e = secs[name]
    return txt[s:e].strip()

def commas_outside_quotes(txt: str):
    # Return list of (index, context) where a comma is outside quotes
    out = []
    in_quotes = False
    for i, ch in enumerate(txt):
        if ch == '"':
            in_quotes = not in_quotes
        if ch == ',' and not in_quotes:
            out.append((i, txt[max(0,i-20):i+20].replace('\n',' ')))
    return out

def check_roadmap_bars(roadmap: str):
    # Allow only uppercase letters, pipes, spaces, parentheses, and keywords
    invalid_ts = re.findall(r'\b\d{1,2}:\d{2}\b', roadmap)
    bad_chars = re.findall(r'[^A-Z\|\s\(\)]', roadmap.replace("CHORUS (repeat)","CHORUS"))
    return invalid_ts, bad_chars

def check_performance(perf: str):
    cues = any(line.strip().startswith(("MIX:", "FX:", "SCENE:")) for line in perf.splitlines())

    headers = re.findall(r'^\[([A-Za-z_\-]+)\]', txt, flags=re.M)

    seq = []
    for h in headers:
        if h in ORDER:

        if idx < last_idx:
            issues.append(f"Out-of-order section: [{h}]")

    # Check required set
